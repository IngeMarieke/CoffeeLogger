# Firestore Rules for Google Authentication
# Copy these rules to Firebase Console → Firestore Database → Rules

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // Users can only access their own data
    match /users/{userId}/logs/{logId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
    }
  }
}

# These rules ensure that:
# 1. Only authenticated users can access data
# 2. Users can only access their own logs (user-specific data)
# 3. Each user's data is isolated from other users
